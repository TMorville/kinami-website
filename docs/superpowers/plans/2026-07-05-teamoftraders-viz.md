```python
"""Export a sanitized showcase.json for the kinami.io teamoftraders visualization.

Reads runs/paper_{eu,us}/ (trade_log.csv, portfolio_history.csv, strategies/,
events.jsonl), computes derived values — FIFO round-trip attribution via
claude_trader.analytics.compute_trade_attribution, equity vs a
currency-normalized benchmark, ISO-week tagging, narrative beats — strips
private fields, validates the schema, and writes one JSON file.

The website consumes only this file; it never touches the private system.
This is the OSS-safe seam (spec §3, website repo
docs/superpowers/specs/2026-07-05-teamoftraders-viz-design.md).

Usage: uv run python -m scripts.export_showcase --out <path/to/showcase.json>
"""

from __future__ import annotations

import argparse
import csv
import json
import re
from datetime import date, datetime, timezone
from pathlib import Path
from typing import Any, cast

from claude_trader.analytics import compute_trade_attribution

RECONCILE_PREFIX = "reconcile_"
WEEK_RE = re.compile(r"(\d{4}-W\d{2})")
REV_RE = re.compile(r"_[vr](\d+)", re.IGNORECASE)

MARKETS: dict[str, dict[str, Any]] = {
    "us": {"dir": "paper_us", "currency": "USD", "starting_capital": 10000,
           "benchmark_label": "QQQ"},
    "eu": {"dir": "paper_eu", "currency": "EUR", "starting_capital": 10000,
           "benchmark_label": "EXS1 (DAX ETF)"},
}


def load_trades(run_dir: Path, market_key: str) -> list[dict[str, Any]]:
    """Parse trade_log.csv → showcase trade dicts (fill_id kept as internal join key)."""
    trades: list[dict[str, Any]] = []
    with open(run_dir / "trade_log.csv") as f:
        for row in csv.DictReader(f):
            fill_id = row.get("fill_id", "")
            if fill_id.startswith(RECONCILE_PREFIX):
                continue  # backfill entries, same rule as compute_trade_attribution
            ts = row["timestamp"]
            iso_year, iso_week, _ = datetime.fromisoformat(ts).isocalendar()
            shares = int(row["shares"])
            price = float(row["price"])
            trades.append({
                "id": f"{market_key}-{len(trades) + 1:04d}",
                "date": ts,
                "ticker": row["ticker"],
                "action": row["action"],
                "shares": shares,
                "price": price,
                "notional": round(shares * price, 2),
                "week_id": f"{iso_year}-W{iso_week:02d}",
                "reasoning": row.get("reasoning", ""),
                "agent": "trader",
                "realized_pnl": None,
                "realized_pnl_pct": None,
                "is_win": None,
                "fill_id": fill_id,  # stripped by sanitize() before output
            })
    return trades


def load_equity(run_dir: Path) -> list[dict[str, Any]]:
    """portfolio_history.csv → daily equity + benchmark normalized to currency.

    History starts mid-run (2026-05-08), so the benchmark is seeded at the first
    equity value: benchmark_equity[i] = equity[0] * bench[i] / bench[0] — "the
    same money in the index from the first common date". (Approved deviation
    from spec §3.3 wording, which assumed full-span history.)
    """
    with open(run_dir / "portfolio_history.csv") as f:
        rows = list(csv.DictReader(f))
    if not rows:
        return []
    equity0 = float(rows[0]["total_equity"])
    bench0 = float(rows[0]["benchmark_close"])
    return [
        {
            "date": row["date"],
            "equity": round(float(row["total_equity"]), 2),
            "benchmark_equity": round(equity0 * float(row["benchmark_close"]) / bench0, 2),
            "daily_pnl": round(float(row["daily_pnl"]), 2),
        }
        for row in rows
    ]


def _revision_rank(filename: str) -> int:
    m = REV_RE.search(filename)
    return int(m.group(1)) if m else 0


def load_weeks(run_dir: Path) -> list[dict[str, Any]]:
    """strategies/*.json → showcase weeks[], newest revision per ISO week.

    Handles both naming families (weekly_strategy_2026-W15.json and
    2026-W21_eu_strategy.json) and revision suffixes (_v2, _r2). Files
    predating watchlist_entries fall back to the legacy string watchlist.
    """
    by_week: dict[str, Path] = {}
    for path in sorted((run_dir / "strategies").glob("*.json")):
        m = WEEK_RE.search(path.name)
        if not m:
            continue
        week_id = m.group(1)
        if week_id not in by_week or _revision_rank(path.name) >= _revision_rank(by_week[week_id].name):
            by_week[week_id] = path
    weeks: list[dict[str, Any]] = []
    for week_id in sorted(by_week):
        raw = json.loads(by_week[week_id].read_text())
        entries = raw.get("watchlist_entries") or [
            {"ticker": t, "direction": None, "thesis": "", "triggers": []}
            for t in raw.get("watchlist", [])
            if isinstance(t, str)
        ]
        weeks.append({
            "week_id": raw.get("week_id", week_id),
            "created_at": raw.get("created_at", ""),
            "generated_by": raw.get("generated_by", ""),
            "market_thesis": raw.get("market_thesis", ""),
            "risk_posture": raw.get("risk_posture", ""),
            "watchlist": [
                {
                    "ticker": e.get("ticker", ""),
                    "direction": e.get("direction"),
                    "thesis": e.get("thesis", ""),
                    "triggers": [
                        {
                            "action": t.get("action", ""),
                            "comparison": t.get("comparison", ""),
                            "price_threshold": t.get("price_threshold"),
                        }
                        for t in e.get("triggers", [])
                    ],
                }
                for e in entries
            ],
            "key_events": [
                {
                    "date": k.get("date", ""),
                    "ticker": k.get("ticker", ""),
                    "severity": k.get("severity", ""),
                    "description": k.get("description", ""),
                }
                for k in raw.get("key_events", [])
            ],
            "notes": raw.get("notes", ""),
            "revisions": [],  # v1: mid-week revision diffs deferred (spec §11)
        })
    return weeks
```

```python
def attach_outcomes(trades: list[dict[str, Any]], run_dir: Path) -> None:
    """Join FIFO round-trips onto trades: is_win on both legs, realized pnl on the exit.

    Reuses the repo's canonical matcher (claude_trader.analytics); trades whose
    fill never closed a round-trip stay is_win=None (open / unmatched).
    """
    attribution = compute_trade_attribution(run_dir / "trade_log.csv", open_positions=[])
    round_trips = cast(list[dict[str, Any]], attribution["round_trips"])
    by_entry: dict[str, dict[str, Any]] = {}
    by_exit: dict[str, dict[str, Any]] = {}
    for rt in round_trips:
        if rt.get("entry_fill_id"):
            by_entry[str(rt["entry_fill_id"])] = rt
        if rt.get("exit_fill_id"):
            by_exit[str(rt["exit_fill_id"])] = rt
    for trade in trades:
        fid = str(trade["fill_id"])
        if not fid:
            continue  # e.g. the DOCN $0 data-error fill has no fill_id
        rt = by_exit.get(fid) or by_entry.get(fid)
        if rt is None:
            continue
        trade["is_win"] = rt["outcome"] == "win"
        if fid == str(rt.get("exit_fill_id")):
            trade["realized_pnl"] = rt["pnl"]
            trade["realized_pnl_pct"] = rt["pnl_pct"]


def build_market(runs_root: Path, market_key: str) -> dict[str, Any]:
    """Assemble one spec-§3.5 market object from its run directory."""
    meta = MARKETS[market_key]
    run_dir = runs_root / str(meta["dir"])
    trades = load_trades(run_dir, market_key)
    attach_outcomes(trades, run_dir)
    return {
        "currency": meta["currency"],
        "starting_capital": meta["starting_capital"],
        "benchmark_label": meta["benchmark_label"],
        "equity": load_equity(run_dir),
        "trades": trades,
        "weeks": load_weeks(run_dir),
    }
```

### Task 3: Beats, sanitization, validation, CLI — and the real export

**Files:**
- Modify: `scripts/export_showcase.py` (append)
- Test: `tests/test_export_showcase.py` (append)

**Interfaces:**
- Consumes: Task 2's `build_market`, `MARKETS`.
- Produces: `derive_beats(markets: dict[str, dict]) -> list[dict]`, `find_context_note(run_dir: Path, beat_date: str) -> str | None`, `sanitize(obj: Any) -> Any`, `assert_sanitized(obj: Any, path: str = "$") -> None`, `validate_showcase(payload: dict) -> None`, `build_showcase(runs_root: Path, generated_at: str) -> dict`, CLI `main() -> int`. The website consumes the emitted `showcase.json`.

**Beat derivation rules (programmatic, deterministic):**
1. `genesis` — earliest trade across markets.
2. `fatfinger` (kind `disaster`) — first trade whose reasoning contains `EMERGENCY` (case-insensitive).
3. `best-win` — exit trade with max `realized_pnl_pct` (percentage is currency-neutral).
4. `worst-drawdown` (kind `drawdown`) — worst peak→trough % on either market's equity; `trade_id: null`.
5. `current-state` (kind `current`) — `market: null`, last equity date, note holds both markets' final equity vs benchmark in raw currency.

Trade beats (1–3) get a `context_note`: nearest same-market `events.jsonl` message within ±1 day, analyst preferred, first 300 chars.

```python
PRIVATE_KEYS = {"fill_id", "entry_fill_id", "exit_fill_id", "cycle_id", "con_id", "isin"}


def derive_beats(markets: dict[str, dict[str, Any]]) -> list[dict[str, Any]]:
    """Deterministic narrative beats — genesis, disaster, best win, drawdown, now."""
    beats: list[dict[str, Any]] = []
    all_trades = [(mk, t) for mk, m in markets.items() for t in m["trades"]]

    mk, first = min(all_trades, key=lambda p: str(p[1]["date"]))
    beats.append({
        "id": "genesis", "kind": "genesis", "market": mk, "trade_id": first["id"],
        "date": str(first["date"])[:10], "title": "First position",
        "note": f"{str(first['action']).upper()} {first['shares']} {first['ticker']} @ {first['price']}",
    })

    for mk, t in sorted(all_trades, key=lambda p: str(p[1]["date"])):
        if "EMERGENCY" in str(t["reasoning"]).upper():
            beats.append({
                "id": "fatfinger", "kind": "disaster", "market": mk, "trade_id": t["id"],
                "date": str(t["date"])[:10], "title": "The fat finger",
                "note": str(t["reasoning"])[:200],
            })
            break

    exits = [(mk, t) for mk, t in all_trades if t.get("realized_pnl_pct") is not None]
    if exits:
        mk, best = max(exits, key=lambda p: float(p[1]["realized_pnl_pct"]))
        beats.append({
            "id": "best-win", "kind": "best-win", "market": mk, "trade_id": best["id"],
            "date": str(best["date"])[:10],
            "title": f"+{best['realized_pnl_pct']:.1f}% on {best['ticker']}",
            "note": str(best["reasoning"])[:200],
        })

    worst: tuple[float, str, str, str] | None = None  # (dd_pct, market, peak_date, trough_date)
    for mk, m in markets.items():
        peak, peak_date = 0.0, ""
        for row in m["equity"]:
            if row["equity"] > peak:
                peak, peak_date = row["equity"], row["date"]
            dd = (row["equity"] - peak) / peak * 100 if peak else 0.0
            if worst is None or dd < worst[0]:
                worst = (dd, mk, peak_date, row["date"])
    if worst and worst[0] < 0:
        dd, mk, peak_date, trough_date = worst
        beats.append({
            "id": "worst-drawdown", "kind": "drawdown", "market": mk, "trade_id": None,
            "date": trough_date, "title": f"{dd:.1f}% drawdown",
            "note": f"Peak {peak_date} → trough {trough_date}.",
        })

    notes, last_date = [], ""
    for mk, m in markets.items():
        if m["equity"]:
            last = m["equity"][-1]
            sym = "€" if m["currency"] == "EUR" else "$"
            notes.append(
                f"{mk.upper()} {sym}{last['equity']:,.0f} vs benchmark "
                f"{sym}{last['benchmark_equity']:,.0f}"
            )
            last_date = max(last_date, last["date"])
    beats.append({
        "id": "current-state", "kind": "current", "market": None, "trade_id": None,
        "date": last_date, "title": "Where it stands", "note": " · ".join(notes),
    })
    return beats


def find_context_note(run_dir: Path, beat_date: str) -> str | None:
    """Nearest agent_log message within ±1 day of the beat; analyst preferred."""
    path = run_dir / "events.jsonl"
    if not path.exists():
        return None
    target = date.fromisoformat(beat_date)
    best: tuple[int, int, str] | None = None  # (day_delta, agent_rank, message)
    with open(path) as f:
        for line in f:
            try:
                ev = json.loads(line)
                d = datetime.fromisoformat(ev["timestamp"]).date()
            except (json.JSONDecodeError, KeyError, ValueError):
                continue
            msg = str(ev.get("message", ""))
            if not msg:
                continue
            delta = abs((d - target).days)
            if delta > 1:
                continue
            rank = 0 if ev.get("data", {}).get("agent") == "analyst" else 1
            if best is None or (delta, rank) < best[:2]:
                best = (delta, rank, msg)
    return best[2][:300] if best else None


def sanitize(obj: Any) -> Any:
    """Recursively drop private keys. Builders shouldn't emit them; belt and braces."""
    if isinstance(obj, dict):
        return {k: sanitize(v) for k, v in obj.items() if k not in PRIVATE_KEYS}
    if isinstance(obj, list):
        return [sanitize(v) for v in obj]
    return obj


def assert_sanitized(obj: Any, path: str = "$") -> None:
    """Raise ValueError on any private key or local filesystem path. A leak is a crash."""
    if isinstance(obj, dict):
        for k, v in obj.items():
            if k in PRIVATE_KEYS:
                raise ValueError(f"private key {k!r} at {path}")
            assert_sanitized(v, f"{path}.{k}")
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            assert_sanitized(v, f"{path}[{i}]")
    elif isinstance(obj, str) and "/Users/" in obj:
        raise ValueError(f"filesystem path leaked at {path}")


def validate_showcase(payload: dict[str, Any]) -> None:
    """Hand-rolled schema contract (spec §3.5) — the promise a live feed must keep."""
    for key in ("generated_at", "span", "markets", "beats"):
        if key not in payload:
            raise ValueError(f"missing top-level key {key!r}")
    for mk in ("eu", "us"):
        market = payload["markets"][mk]
        for key in ("currency", "starting_capital", "benchmark_label",
                    "equity", "trades", "weeks"):
            if key not in market:
                raise ValueError(f"markets.{mk} missing {key!r}")
        for trade in market["trades"]:
            for key in ("id", "date", "ticker", "action", "shares", "price", "notional",
                        "week_id", "reasoning", "agent", "realized_pnl",
                        "realized_pnl_pct", "is_win"):
                if key not in trade:
                    raise ValueError(f"trade {trade.get('id')} missing {key!r}")
        for week in market["weeks"]:
            for key in ("week_id", "market_thesis", "risk_posture",
                        "watchlist", "key_events", "notes"):
                if key not in week:
                    raise ValueError(f"week {week.get('week_id')} missing {key!r}")
    for beat in payload["beats"]:
        for key in ("id", "kind", "market", "trade_id", "date", "title", "note"):
            if key not in beat:
                raise ValueError(f"beat {beat.get('id')} missing {key!r}")


def build_showcase(runs_root: Path, generated_at: str) -> dict[str, Any]:
    """Assemble, sanitize, and validate the full showcase payload."""
    markets = {mk: build_market(runs_root, mk) for mk in MARKETS}
    beats = derive_beats(markets)
    for beat in beats:
        if beat["market"] and beat["trade_id"]:
            note = find_context_note(
                runs_root / str(MARKETS[beat["market"]]["dir"]), beat["date"])
            if note:
                beat["context_note"] = note
    equity_dates = [r["date"] for m in markets.values() for r in m["equity"]]
    trade_dates = [str(t["date"])[:10] for m in markets.values() for t in m["trades"]]
    payload: dict[str, Any] = {
        "generated_at": generated_at,
        "span": {"start": min(trade_dates + equity_dates),
                 "end": max(trade_dates + equity_dates)},
        "markets": markets,
        "beats": beats,
    }
    payload = sanitize(payload)
    assert_sanitized(payload)
    validate_showcase(payload)
    return payload


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--runs-root", type=Path, default=Path("runs"))
    parser.add_argument("--out", type=Path, default=Path("showcase.json"))
    args = parser.parse_args()
    payload = build_showcase(args.runs_root, datetime.now(timezone.utc).isoformat())
    args.out.parent.mkdir(parents=True, exist_ok=True)
    args.out.write_text(json.dumps(payload, indent=1))
    print(f"wrote {args.out} ({args.out.stat().st_size:,} bytes)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```js
// Pure data layer for the teamoftraders viz: indexing, formatting, hashing.
// No DOM, no fetch — node-testable. The raw payload is loaded by main.js.

/** Index the showcase payload for O(1) lookups and a shared day axis. */
export function indexShowcase(raw) {
  const tradesById = new Map();
  const weeksByKey = new Map();
  const tradesByWeek = new Map();
  const days = new Set();
  for (const market of ['eu', 'us']) {
    const m = raw.markets[market];
    for (const trade of m.trades) {
      trade.market = market;
      tradesById.set(trade.id, trade);
      days.add(trade.date.slice(0, 10));
      const wk = `${market}:${trade.week_id}`;
      if (!tradesByWeek.has(wk)) tradesByWeek.set(wk, []);
      tradesByWeek.get(wk).push(trade);
    }
    for (const row of m.equity) days.add(row.date);
    for (const week of m.weeks) weeksByKey.set(`${market}:${week.week_id}`, week);
  }
  const dayList = [...days].sort();
  const dayIndex = new Map(dayList.map((d, i) => [d, i]));
  return { raw, tradesById, weeksByKey, tradesByWeek, dayList, dayIndex };
}

/** Raw-currency formatter — € / $ headline numbers (spec: raw money, not index points). */
export function fmtMoney(value, currency) {
  return new Intl.NumberFormat(currency === 'EUR' ? 'de-DE' : 'en-US', {
    style: 'currency', currency, maximumFractionDigits: 0,
  }).format(value);
}

/** FNV-1a → [0,1). Deterministic layout jitter — no Math.random anywhere. */
export function hash01(str) {
  let h = 2166136261;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return (h >>> 0) / 4294967296;
}
```

```js
// Pure strata layout: trades → grains in a market column; day rows tile weeks.
// Deterministic (hash01 jitter) so narrative and explorer render identically.
import { hash01 } from './data.js';

export const GRAIN_MIN_R = 1.5;
export const GRAIN_MAX_R = 9;

/** Lay out one market's trades as grains inside a column. */
export function layoutMarket(index, market, { x0, width, rowH }) {
  const trades = index.raw.markets[market].trades;
  const maxNotional = Math.max(...trades.map(t => t.notional), 1);
  return trades.map(t => {
    const day = index.dayIndex.get(t.date.slice(0, 10)) ?? 0;
    return {
      id: t.id,
      day,
      x: x0 + (0.1 + 0.8 * hash01(t.id)) * width,
      y: (day + 0.5) * rowH,
      r: GRAIN_MIN_R + Math.sqrt(t.notional / maxNotional) * (GRAIN_MAX_R - GRAIN_MIN_R),
      ring: t.action === 'sell', // sells carve — rendered as rings, buys deposit as dots
      win: t.is_win,             // true | false | null (open / unmatched)
      phase: hash01(t.id + 'p') * Math.PI * 2,
    };
  });
}

/** Contiguous ISO-week bands over the shared day axis. */
export function weekBands(index, rowH) {
  const bands = [];
  let current = null;
  index.dayList.forEach((d, i) => {
    const week = isoWeekId(d);
    if (!current || current.week !== week) {
      current = { week, y0: i * rowH, y1: (i + 1) * rowH };
      bands.push(current);
    } else {
      current.y1 = (i + 1) * rowH;
    }
  });
  return bands;
}

/** ISO-8601 week id for a YYYY-MM-DD string (UTC math, no locale). */
export function isoWeekId(dateStr) {
  const d = new Date(dateStr + 'T00:00:00Z');
  const day = (d.getUTCDay() + 6) % 7;         // Mon=0 … Sun=6
  d.setUTCDate(d.getUTCDate() - day + 3);      // nearest Thursday decides the year
  const jan4 = new Date(Date.UTC(d.getUTCFullYear(), 0, 4));
  const week = 1 + Math.round(
    ((d - jan4) / 86400000 - 3 + ((jan4.getUTCDay() + 6) % 7)) / 7);
  return `${d.getUTCFullYear()}-W${String(week).padStart(2, '0')}`;
}
```

```js
// Canvas2D renderer for the strata field. One code path serves both the
// narrative (revealDay scrubbed by scroll) and the explorer (fully revealed,
// filtered/highlighted by interaction). Visual identity: single-hue amber
// grains on void, additive glow, whisper week lines.

const WIN = a => `rgba(240, 200, 130, ${a})`;
const AMBER = a => `rgba(220, 180, 100, ${a})`;

export function createRenderer(canvas, { grainsByMarket, bands, columns }) {
  const ctx = canvas.getContext('2d');
  let dpr = 1;

  function resize() {
    dpr = window.devicePixelRatio || 1;
    const { width, height } = canvas.getBoundingClientRect();
    canvas.width = Math.round(width * dpr);
    canvas.height = Math.round(height * dpr);
  }

  function grainColor(g, alphaScale = 1) {
    if (g.win === true) return WIN(0.85 * alphaScale);
    if (g.win === false) return AMBER(0.30 * alphaScale);
    return AMBER(0.55 * alphaScale);
  }

  function draw({ revealDay = Infinity, t = 0, highlightId = null, filter = null } = {}) {
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, canvas.width / dpr, canvas.height / dpr);
    const w = canvas.width / dpr;

    // Week bands — whisper dividers + mono labels in the center gutter.
    ctx.font = '300 10px "DM Mono", monospace';
    ctx.textAlign = 'center';
    for (const band of bands) {
      ctx.strokeStyle = AMBER(0.06);
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(0, band.y0 + 0.5);
      ctx.lineTo(w, band.y0 + 0.5);
      ctx.stroke();
      if (band.y0 <= (revealDay + 1) * columns.rowH) {
        ctx.fillStyle = AMBER(0.35);
        ctx.fillText(band.week, w / 2, band.y0 + 12);
      }
    }

    // Grains — additive glow; reveal follows the scrubbed day.
    ctx.globalCompositeOperation = 'lighter';
    for (const market of ['eu', 'us']) {
      for (const g of grainsByMarket[market]) {
        if (g.day > revealDay) continue;
        const dimmed = filter && !filter(g);
        const alphaScale = dimmed ? 0.15 : 1;
        const y = g.y + Math.sin(t * 0.0003 + g.phase) * 1.2;

        ctx.fillStyle = grainColor(g, 0.25 * alphaScale); // halo
        ctx.beginPath();
        ctx.arc(g.x, y, g.r * 2.2, 0, Math.PI * 2);
        ctx.fill();

        if (g.ring) {
          ctx.strokeStyle = grainColor(g, alphaScale);
          ctx.lineWidth = 1.5;
          ctx.beginPath();
          ctx.arc(g.x, y, g.r, 0, Math.PI * 2);
          ctx.stroke();
        } else {
          ctx.fillStyle = grainColor(g, alphaScale);
          ctx.beginPath();
          ctx.arc(g.x, y, g.r, 0, Math.PI * 2);
          ctx.fill();
        }

        if (g.id === highlightId) {
          ctx.strokeStyle = WIN(0.9);
          ctx.lineWidth = 1;
          ctx.beginPath();
          ctx.arc(g.x, y, g.r + 5, 0, Math.PI * 2);
          ctx.stroke();
        }
      }
    }
    ctx.globalCompositeOperation = 'source-over';

    // Column captions.
    ctx.font = '300 10px "DM Mono", monospace';
    ctx.fillStyle = AMBER(0.55);
    ctx.textAlign = 'center';
    ctx.fillText('EU · XETRA', columns.eu.x0 + columns.eu.width / 2, 16);
    ctx.fillText('US · NYSE/NASDAQ', columns.us.x0 + columns.us.width / 2, 16);
  }

  resize();
  return { draw, resize };
}

/** Shared column geometry: EU left third, US right third, gutter between. */
export function columnGeometry(width, height, dayCount) {
  return {
    eu: { x0: width * 0.06, width: width * 0.36 },
    us: { x0: width * 0.58, width: width * 0.36 },
    rowH: (height - 40) / Math.max(dayCount, 1),
  };
}
```

```js
// trace's x-deviation from the column center is performance vs the first
// snapshot. Benchmark rides alongside, dashed — same money in the index.

/** SVG path for one field of an equity series. Pure — node-testable. */
export function corePath(rows, dayIndex, { xCenter, xScale, rowH }, field = 'equity') {
  if (!rows.length) return '';
  const base = rows[0][field];
  return rows
    .map((row, i) => {
      const x = xCenter + (row[field] / base - 1) * xScale;
      const y = ((dayIndex.get(row.date) ?? 0) + 0.5) * rowH;
      return `${i ? 'L' : 'M'}${x.toFixed(1)} ${y.toFixed(1)}`;
    })
    .join(' ');
}
```

```js
// Pure narrative structure: merge weekly-strategy chapters and curated beats
// into an ordered list of scroll stops. No gsap, no DOM — node-testable.
import { isoWeekId } from './strata.js';

/** HTML-escape untrusted showcase prose before innerHTML. */
export function esc(s) {
  return String(s).replace(/[&<>"']/g, c => (
    { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
}

/** Chapters (weeks that have a strategy in either market) + beats, by day. */
export function buildStops(index) {
  const stops = [];
  const seen = new Set();
  index.dayList.forEach((d, dayIdx) => {
    const week = isoWeekId(d);
    if (seen.has(week)) return;
    seen.add(week);
    const eu = index.weeksByKey.get(`eu:${week}`);
    const us = index.weeksByKey.get(`us:${week}`);
    if (eu || us) stops.push({ kind: 'chapter', week, day: dayIdx, eu, us });
  });
  for (const beat of index.raw.beats) {
    const day = index.dayIndex.get(beat.date) ?? index.dayList.length - 1;
    stops.push({ kind: 'beat', beat, day });
  }
  stops.sort((a, b) => a.day - b.day || (a.kind === 'chapter' ? -1 : 1));
  return stops;
}
```

```js
// GSAP narrative pass: pinned stage, scrubbed strata reveal, equity draw-on,
// chapter/beat cards with SplitText inscriptions. Browser-only module —
// main.js dynamic-imports it so static mode never loads gsap.
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import { SplitText } from 'gsap/SplitText';
import { buildStops, esc } from './narrative.js';

gsap.registerPlugin(ScrollTrigger, SplitText);

function chapterHTML(stop) {
  const thesis = (label, w) => w
    ? `<blockquote>${label} — ${esc(w.market_thesis.slice(0, 160))}…</blockquote>` : '';
  return `
    <span class="label agent-tag">${stop.week} · strategist</span>
    <div class="card-title">the plan</div>
    ${thesis('EU', stop.eu)}${thesis('US', stop.us)}`;
}

function beatHTML(beat) {
  const who = beat.market ? beat.market.toUpperCase() : 'BOTH BOOKS';
  return `
    <span class="label agent-tag">${beat.date} · ${who} · trader</span>
    <div class="card-title">${esc(beat.title)}</div>
    <blockquote>${esc(beat.note)}</blockquote>
    ${beat.context_note
      ? `<blockquote class="label">context · ${esc(beat.context_note.slice(0, 180))}…</blockquote>`
      : ''}`;
}

export function initNarrative(index, equityPaths, cardsEl, spacerEl) {
  const stops = buildStops(index);
  const dayCount = index.dayList.length;
  spacerEl.style.height = `${stops.length * 80}vh`;

  // Equity draw-on: dash the solid paths to their own length, scrub the offset.
  const dashTargets = [];
  for (const market of ['eu', 'us']) {
    const path = equityPaths[market].equity;
    const len = path.getTotalLength();
    path.style.strokeDasharray = `${len}`;
    path.style.strokeDashoffset = `${len}`;
    dashTargets.push({ path, len });
  }

  const state = { revealDay: 0 };
  ScrollTrigger.create({
    trigger: '#narrative', start: 'top top', end: 'bottom bottom', scrub: 0.5,
    onUpdate(self) {
      state.revealDay = self.progress * dayCount;
      for (const { path, len } of dashTargets) {
        path.style.strokeDashoffset = `${(1 - self.progress) * len}`;
      }
    },
  });

  stops.forEach((stop, i) => {
    const el = document.createElement('div');
    el.className = 'card';
    el.innerHTML = stop.kind === 'chapter' ? chapterHTML(stop) : beatHTML(stop.beat);
    cardsEl.appendChild(el);

    const start = i / stops.length;
    const dur = 1 / stops.length;
    const tl = gsap.timeline({
      scrollTrigger: { trigger: '#narrative', start: 'top top', end: 'bottom bottom', scrub: true },
    });
    tl.to(el, { opacity: 1, duration: dur * 0.25 }, start + dur * 0.1)
      .to(el, { opacity: 0, duration: dur * 0.25 }, start + dur * 0.7);
    const quote = el.querySelector('blockquote');
    if (quote) {
      const split = new SplitText(quote, { type: 'chars' });
      tl.from(split.chars,
        { opacity: 0.08, stagger: dur * 0.0015, duration: dur * 0.2 }, start + dur * 0.12);
    }
  });

  return state;
}
```

```js
/** Plan-vs-execution join for one market-week: what the strategy intended
 *  vs what actually traded, with realized P&L per planned ticker. */
export function weekExecution(week, trades) {
  const rows = week.watchlist.map(wi => {
    const hits = trades.filter(t => t.ticker === wi.ticker);
    return {
      ticker: wi.ticker,
      direction: wi.direction,
      tradeCount: hits.length,
      realizedPnl: hits.reduce((sum, t) => sum + (t.realized_pnl || 0), 0),
    };
  });
  const planned = new Set(week.watchlist.map(wi => wi.ticker));
  const offPlan = [...new Set(trades.filter(t => !planned.has(t.ticker)).map(t => t.ticker))];
  return { rows, offPlan };
}
```

```js
    `<button role="tab" data-week-id="${w}">${w.replace('2026-', '')}</button>`).join('');
  rail.addEventListener('click', ev => {
    const btn = ev.target.closest('button');
    if (!btn) return;
    const w = btn.dataset.weekId;
    const market = index.weeksByKey.has(`eu:${w}`) ? 'eu' : 'us';
    api.openStrategyPanel(`${market}:${w}`);
  });

  api.openStrategyPanel = key => {
    const [market, weekId] = key.split(':');
    const week = index.weeksByKey.get(key);
    if (!week) return;
    const m = index.raw.markets[market];
    const trades = index.tradesByWeek.get(key) || [];
    const { rows, offPlan } = weekExecution(week, trades);
    const other = market === 'eu' ? 'us' : 'eu';
    const otherKey = `${other}:${weekId}`;
    const execRows = rows.map(r => `
      <tr data-ticker="${esc(r.ticker)}"><td>${esc(r.ticker)}</td><td>${r.direction ?? '—'}</td>
        <td>${r.tradeCount
          ? `${r.tradeCount} trade${r.tradeCount > 1 ? 's' : ''} · ${fmtMoney(r.realizedPnl, m.currency)}`
          : 'untouched'}</td></tr>`).join('');
    strategyPanel.innerHTML = `
      <button class="close" aria-label="close">×</button>
      <h2>${weekId} <span class="label">${market} · ${esc(week.risk_posture || '')}</span></h2>
      ${index.weeksByKey.has(otherKey)
        ? `<button class="pill" data-switch="${otherKey}">view ${other} week →</button>` : ''}
      <div class="section"><span class="label">market thesis · strategist</span>
        <blockquote>${esc(week.market_thesis)}</blockquote></div>
      <div class="section"><span class="label">plan vs execution — click a row to filter</span>
        <table><tr><th>ticker</th><th>intent</th><th>what happened</th></tr>${execRows}</table>
        ${offPlan.length
          ? `<p class="label">off-plan trades: ${offPlan.map(esc).join(', ')}</p>` : ''}</div>
      ${week.key_events.length ? `
        <div class="section"><span class="label">key events</span>
          <table>${week.key_events.map(k => `
            <tr><td>${esc(k.date)}</td><td>${esc(k.ticker)}</td><td>${esc(k.description)}</td></tr>`).join('')}
          </table></div>` : ''}`;
    strategyPanel.classList.add('open');
    strategyPanel.querySelector('.close').onclick = () => strategyPanel.classList.remove('open');
    strategyPanel.querySelectorAll('tr[data-ticker]').forEach(tr => {
      tr.style.cursor = 'pointer';
      tr.onclick = () => {
        filterState.ticker = tr.dataset.ticker;
        applyFilter();
        strategyPanel.classList.remove('open');
      };
    });
    const switchBtn = strategyPanel.querySelector('[data-switch]');
    if (switchBtn) switchBtn.onclick = () => api.openStrategyPanel(switchBtn.dataset.switch);
  };

  // ---- crosshair readout ----------------------------------------------
  const readout = document.getElementById('crosshair-readout');
  const equityByDate = { eu: new Map(), us: new Map() };
  for (const market of ['eu', 'us']) {
    for (const row of index.raw.markets[market].equity) equityByDate[market].set(row.date, row);
  }
  canvas.addEventListener('pointermove', ev => {
    const [, y] = localXY(ev);
    const date = index.dayList[Math.max(0, Math.min(
      Math.floor(y / rowH), index.dayList.length - 1))];
    const parts = [date];
    for (const market of ['eu', 'us']) {
      const row = equityByDate[market].get(date);
      if (row) {
        const cur = index.raw.markets[market].currency;
        parts.push(`${market} ${fmtMoney(row.equity, cur)} vs ${fmtMoney(row.benchmark_equity, cur)}`);
      }
    }
    readout.textContent = parts.join(' · ');
  });
```
